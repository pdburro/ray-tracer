cmake_minimum_required(VERSION 3.16)
project(RayTracer LANGUAGES CXX)

add_library(texture_image src/image.cpp)
target_compile_features(texture_image PUBLIC cxx_std_17)
target_include_directories(texture_image SYSTEM PRIVATE external)

add_executable(ray_tracer src/main.cpp)
target_link_libraries(ray_tracer PRIVATE texture_image)
target_compile_features(ray_tracer PRIVATE cxx_std_17)
set_target_properties(ray_tracer PROPERTIES CXX_EXTENSIONS OFF)
add_custom_command(TARGET ray_tracer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:ray_tracer>/assets")

if(MSVC)
    target_compile_options(ray_tracer PRIVATE /W4)
else()
    target_compile_options(ray_tracer PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)

    target_include_directories(intersection_tests PRIVATE src)
    target_compile_features(intersection_tests PRIVATE cxx_std_17)

    add_executable(diffuse_tests tests/diffuse.cpp)
    target_include_directories(diffuse_tests PRIVATE src)
    target_compile_features(diffuse_tests PRIVATE cxx_std_17)
    add_test(NAME diffuse COMMAND diffuse_tests)
    add_executable(material_tests tests/materials.cpp)
    target_include_directories(material_tests PRIVATE src)
    target_compile_features(material_tests PRIVATE cxx_std_17)
    add_test(NAME materials COMMAND material_tests)
    add_executable(camera_tests tests/camera.cpp)
    target_include_directories(camera_tests PRIVATE src)
    target_compile_features(camera_tests PRIVATE cxx_std_17)
    add_test(NAME camera COMMAND camera_tests)
    add_executable(motion_tests tests/motion.cpp)
    target_include_directories(motion_tests PRIVATE src)
    target_compile_features(motion_tests PRIVATE cxx_std_17)
    add_test(NAME motion COMMAND motion_tests)
    add_executable(bvh_tests tests/bvh.cpp)
    target_include_directories(bvh_tests PRIVATE src)
    target_compile_features(bvh_tests PRIVATE cxx_std_17)
    add_test(NAME bvh COMMAND bvh_tests)
